###### 数据传递

首先在App.vue中创建一个数据向子传递和创建一个方法传递给子组件，具体实现如下：

App.vue的源码如下：

```vue
<template>
  <div id="app">
    <div class="todo-page">
      <div class="todo-container">
        <UserHeader :addTodo="addTodo" />
        <UserLists :todos="todos" />
        <UserFooter />
      </div>
    </div>
  </div>
</template>
<script>
import UserHeader from "./components/UserHeader";
import UserFooter from "./components/UserFooter";
import UserLists from "./components/UserLists";
export default {
  name: "App",
  components: { UserHeader, UserFooter, UserLists },
  data() {
    return {
      todos: [
        { id: "001", title: "抽烟", done: true },
        { id: "002", title: "喝酒", done: true },
        { id: "003", title: "烫头", done: true },
        { id: "004", title: "开车", done: false }
      ]
    };
  },
  methods: {
    addTodo(todoObj) {
      if (!todoObj) {
        alert("输入数据不能为空");
        return;
      }
      this.todos.unshift(todoObj);
    }
  }
};
</script>
```

> 1. addTodo方法传递给子组件，用于将创建的待完成事项传递回来
> 2. todos是所有待完成事项的模拟数据数组

UserHeader.vue的源码如下：

```vue
<template>
  <!-- 头部显示 -->
  <div class="todo-header">
    <input
      type="text"
      placeholder="请输入您的任务名称，按回车确认"
      v-model="title"
      @keyup.enter="add"
    />
  </div>
</template>
<script>
import { nanoid } from "nanoid";
export default {
  name: "UserHeader",
  data() {
    return { title: "" };
  },
  methods: {
    //当用户点击enter键之后，进入此方式的执行
    add() {
      if (!this.title.trim()) return;
      //创建数据对象todoObj
      const todoObj = { id: nanoid(), title: this.title, done: false };
      //使用父组件传递过来的方法实现数据的子传父功能
      this.addTodo(todoObj);
      //清空显示框数据
      this.title = "";
    }
  },
  props: ["addTodo"]
};
</script>
```

> add方法实际是用户点击回车触发的添加事件，然后对父组件传递过来的addTodo进行实现

##### 事件总线

如下为通过全局事件总线+axios请求方式实现：
UserSearch.vue:

```vue
<template>
  <div class="container">
    <section class="jumbotron">
      <h3 class="jumbotron-heading">Search Github Users</h3>
      <div>
        <input
          type="text"
          placeholder="enter the name you search"
          v-model="info.keyWord"
        />&nbsp;<button @click="getUsers">Search</button>
      </div>
    </section>
  </div>
</template>
<script>
import axios from "axios";
export default {
  name: "UserSearch",
  data() {
    return {
      info: {
        keyWord: ""
      }
    };
  },
  methods: {
    // 点击按钮后的数据展示
    getUsers() {
      //请求前更新List的数据
      this.$bus.$emit("updateData", {
        isFirst: false,
        isLoading: true,
        errMsg: "",
        users: []
      });
      axios
        .get(`https://api.github.com/search/users?q=${this.info.keyWord}`)
        .then(
          (response) => {
            //请求成功后更新List的数据
            this.$bus.$emit("updateData", {
              isLoading: false,
              errMsg: "",
              users: response.data.items
            });
          },
          (error) => {
            //请求失败后更新List的数据
            this.$bus.$emit("updateData", {
              isLoading: false,
              errMsg: error.message,
              users: []
            });
          }
        );
    }
  }
};
</script>
```
